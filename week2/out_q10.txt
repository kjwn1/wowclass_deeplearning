selected_optimizer = optimizer_examples["Adam"]

    metrics=["accuracy"],
)

fashion_history = fashion_model.fit(

    train_target,
    validation_split=0.2,
    epochs=3,
    batch_size=128,
    verbose=1,
)

test_loss, test_accuracy = fashion_model.evaluate(

    test_target,
    verbose=0,
)
print("test accuracy:", test_accuracy)
